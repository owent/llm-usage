import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Token and quota statistics" icon="chart">
    Input, output and cache usage are kept separate. Missing values remain unknown,
    and subscription quotas are shown independently of tokens.
  </Card>
  <Card title="Local storage" icon="shield-check">
    Read local agent files and explicitly enabled local telemetry. No account login,
    cloud database or remote usage report is required.
  </Card>
  <Card title="Collection status" icon="magnifier">
    Collection status explains versions, incomplete fields and parsing failures.
    A supported parser does not imply that every client records every request.
  </Card>
  <Card title="Development references" icon="document">
    Developers can inspect the data model, adapter sources, migrations and native
    acceptance records alongside the implementation.
  </Card>
</CardGrid>
